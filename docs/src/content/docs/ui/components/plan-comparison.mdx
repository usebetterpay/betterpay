## Usage

```tsx title="Import"
import { PlanComparison, type ComparisonFeatureRow } from '@betterpay/ui';
```

```tsx title="Derived from plan.features"
<PlanComparison plans={plans} interval="month" onSelectPlan={handleSelect} />
```

```tsx title="Explicit matrix"
const features: ComparisonFeatureRow[] = [
  {
    id: 'msg',
    label: 'Messages / month',
    values: { free: '100', pro: '5.000', enterprise: 'Unlimited' },
  },
  {
    id: 'wa',
    label: 'WhatsApp',
    values: { free: false, pro: true, enterprise: true },
  },
];

<PlanComparison plans={plans} features={features} interval="year" />
```

## Props

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `plans` | `PlanView[]` | — | Column headers + prices |
| `features` | `ComparisonFeatureRow[]` | derived | Matrix rows |
| `interval` | `"month" \| "year"` | `"month"` | Price period |
| `onSelectPlan` | `(planId) => void` | — | Header CTA |
| `currencyFallback` | `string` | `"IDR"` | Fallback currency |

### ComparisonCell

`true` · `false` · `"partial"` · `string`
